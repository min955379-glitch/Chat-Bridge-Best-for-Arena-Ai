## background.js

<a id="backgroundjs"></a>

*624 lines.*

````js
// background.js — the service worker that runs the loop,
// the "brain" that makes two chats talk to each other.
// It is a Manifest V3 service worker, so it can be stopped by Chrome when it
// looks idle; the 1-second polling and a small heartbeat keep it alive.
// Plain JavaScript. No frameworks.

importScripts('common.js', 'extract.js', 'api-client.js');

const Common = self.ChatBridgeCommon;
const Extract = self.ChatBridgeExtract;
const ChatApi = self.ChatBridgeApi;

// How often the reply is read. The brief asks for 1 second; the tests shrink
// it with the __chatBridgePollMs hook so they finish quickly.
const POLL_MS = (typeof self !== 'undefined' && self.__chatBridgePollMs) || 1000;
// Chrome stops an idle service worker after 30 seconds. Every extension API
// call resets that timer, so this heartbeat (plus the 1-second tab polling)
// keeps the worker alive for the whole run.
const HEARTBEAT_MS = 10000;
const START_TEXT_MAX = 20000;   // do not forward more than this many characters

// The run that is going on right now, or null.
let activeRun = null;
// Storage writes are chained so two writes never overwrite each other.
let writeChain = Promise.resolve();

// ---------- storage helpers ----------
function queueWrite(task) {
  writeChain = writeChain.then(task, task);
  return writeChain;
}

async function getSettings() {
  const stored = await chrome.storage.local.get('settings');
  return Common.normalizeSettings(stored.settings);
}

async function getRoles() {
  const stored = await chrome.storage.local.get('roles');
  const roles = stored.roles || {};
  return { A: roles.A || null, B: roles.B || null };
}

async function setStatus(text, tone, extra) {
  const status = Object.assign({
    text: text,
    tone: tone || 'idle',
    running: Boolean(activeRun),
    updatedAt: Date.now()
  }, extra || {});
  await queueWrite(() => chrome.storage.local.set({ status: status }));
}

async function addLog(text) {
  const line = { t: Date.now(), text: String(text) };
  await queueWrite(async () => {
    const stored = await chrome.storage.local.get('log');
    const list = Array.isArray(stored.log) ? stored.log : [];
    list.push(line);
    await chrome.storage.local.set({ log: list.slice(-20) });
  });
}

async function setRunFlag(running, note) {
  await queueWrite(() => chrome.storage.local.set({
    run: { running: Boolean(running), note: note || '', changedAt: Date.now() }
  }));
}

// ---------- talking to a tab ----------
async function injectContentScripts(tabId) {
  // The page was opened before the extension was installed, so the content
  // script is missing. Put it in now. The files are listed in order.
  await chrome.scripting.executeScript({
    target: { tabId: tabId },
    files: ['selectors.js', 'extract.js', 'common.js', 'content.js']
  });
}

function plainError(err, fallback) {
  const message = err && err.message ? String(err.message) : '';
  if (/Cannot access|chrome:\/\/|edge:\/\/|about:|chrome-extension:/i.test(message)) {
    return 'Chrome does not allow the extension on this page (' + message + '). Open a normal website tab.';
  }
  if (/Receiving end does not exist|Could not establish connection|message port closed/i.test(message)) {
    return 'The page is not ready yet. Open the tab and reload it (F5), then try again.';
  }
  return message || fallback || 'Something went wrong.';
}

// Send one message to a tab. If the content script is missing, add it and retry.
async function sendToTab(tabId, message) {
  try {
    return await chrome.tabs.sendMessage(tabId, message);
  } catch (firstError) {
    try {
      await injectContentScripts(tabId);
    } catch (injectError) {
      throw new Error(plainError(injectError, 'I could not reach the page. Reload the tab and try again.'));
    }
    try {
      return await chrome.tabs.sendMessage(tabId, message);
    } catch (secondError) {
      throw new Error(plainError(secondError, 'The page is not answering. Reload the tab and try again.'));
    }
  }
}

// Read the last reply in a tab: { ok, count, text, code }
// An empty chat is a normal reading (count 0), not an error: a brand-new chat
// must still be usable as Tab A or Tab B. Only a missing pick fails here.
async function readTab(tabId) {
  const reply = await sendToTab(tabId, { type: 'readReply' });
  if (!reply) throw new Error('The page did not answer. Reload the tab and try again.');
  if (reply.error) throw new Error(reply.error);
  // Trust nothing from the page: keep only text and a number.
  const count = Number(reply.count) || 0;
  const text = typeof reply.text === 'string' ? reply.text : String(reply.text == null ? '' : reply.text);
  return {
    ok: true,
    count: count,
    text: count === 0 ? '' : text,
    code: count === 0 ? '' : (typeof reply.code === 'string' ? reply.code : '')
  };
}

// Two pieces of text are "the same" when only spacing and line breaks differ.
function normaliseForCompare(text) {
  return String(text == null ? '' : text).replace(/\s+/g, ' ').trim();
}

// Did the tab's newest reply change into something we did not just send?
// count 0 means "the page shows no reply at all", which is not our message
// and is not the AI answering either.
function isForeignChange(current, sentText) {
  if (current.count === 0) return false;
  const now = normaliseForCompare(current.text);
  if (!now) return false;
  // Our own pasted message (or a piece of it) appeared as the newest reply:
  // wait for the AI's answer instead of forwarding our own text back.
  const sent = normaliseForCompare(sentText);
  if (sent) {
    if (now === sent) return false;
    if (now.startsWith(sent)) return false;
  }
  return true;
}

// Put a message into a tab and press send.
async function deliverToTab(tabId, text) {
  const sent = await sendToTab(tabId, { type: 'deliver', text: text });
  if (!sent) throw new Error('The page did not answer. Reload the tab and try again.');
  if (sent.error) throw new Error(sent.error);
}

// ---------- run control ----------
class StopSignal extends Error {
  constructor() { super('Stopped by you.'); this.name = 'StopSignal'; }
}

function checkStop() {
  if (!activeRun || activeRun.stopRequested) throw new StopSignal();
}

// Wait that reacts to Stop quickly.
async function pause(ms) {
  const step = 100;
  let left = ms;
  while (left > 0) {
    checkStop();
    const slice = Math.min(step, left);
    await new Promise((resolve) => setTimeout(resolve, slice));
    left -= slice;
  }
}

function roleTitle(run, role) {
  const tabId = run.roles[role];
  const title = run.titles[tabId] || ('Tab ' + role);
  return role + ' (' + title + ')';
}

// Choose the part of a reply the user asked for.
function applyMode(reply, mode, markers) {
  const full = Extract.cleanText(reply.text || '');
  if (!full) return { error: 'The last reply is empty right now.' };
  if (mode === 'code') {
    if (reply.code) {
      const codeText = Extract.cleanText(reply.code);
      return codeText ? { text: codeText } : { error: 'The last code block in the reply is empty.' };
    }
    return Extract.applyModeToText(full, 'code', markers);
  }
  if (mode === 'markers') return Extract.betweenMarkers(full, markers);
  return { text: full };
}

// Read the newest reply with the chosen copy mode (used for the first message).
async function readPickedText(run, role) {
  const tabId = run.roles[role];
  const mode = role === 'A' ? run.settings.copyModeA : run.settings.copyModeB;
  const reply = await readTab(tabId);
  const picked = applyMode(reply, mode, { start: run.settings.markerStart, end: run.settings.markerEnd });
  if (picked.error) {
    throw new Error('I could not read the first message from ' + roleTitle(run, role) + ': ' + picked.error);
  }
  return picked.text;
}

// Wait until the tab produces a NEW reply and stops changing.
async function waitForNewReply(run, role, baseline, sentText) {
  const tabId = run.roles[role];
  const silence = run.settings.silence;
  const deadline = Date.now() + run.settings.timeout * 1000;
  let seen = false;
  let lastText = '';
  let lastChange = 0;
  let lastWritten = '';
  // The message we pasted, so we do not mistake it for the AI's answer.
  const sent = normaliseForCompare(sentText);

  for (;;) {
    checkStop();
    const reply = await readTab(tabId);
    const now = Date.now();

    // A change compared with what was on the page before we pasted, that is
    // also not our own message coming back.
    const changed = reply.count !== baseline.count || reply.text !== baseline.text;

    if (!seen && changed && isForeignChange(reply, sent)) {
      seen = true;
      lastText = reply.text;
      lastChange = now;
      await addLog(roleTitle(run, role) + ' started answering.');
    } else if (seen) {
      // While waiting, ignore a return to "no reply" and an echo of our own
      // message: only real text from the AI restarts the silence counter.
      if (isForeignChange(reply, sent) && reply.text !== lastText) {
        lastText = reply.text;
        lastChange = now;
      }
    }

    // Whole seconds with no change in the reply.
    const silentFor = seen ? Math.floor((now - lastChange) / 1000) : 0;
    const unfinished = seen && Extract.looksUnfinished(lastText);
    const needed = unfinished ? silence * 2 : silence;

    const statusText = seen
      ? 'waiting for ' + role + ' to finish (' + silentFor + 's of ' + needed + 's silence)'
      : 'waiting for ' + role + ' to start answering';
    if (statusText !== lastWritten) {
      lastWritten = statusText;
      await setStatus(run.statusPrefix + statusText, 'running', { waitSince: seen ? lastChange : now });
    }

    if (seen && silentFor >= needed && String(lastText || '').trim()) {
      // Return the reading that ended the wait, so a reply that finished at
      // the same moment is not missed.
      return reply.count > 0 ? reply : Object.assign({}, reply, { count: 1, text: lastText });
    }

    if (now > deadline) {
      const hint = seen
        ? 'It started answering but never went quiet. You can raise "seconds of silence" in Settings.'
        : 'It did not start answering. Some sites do not accept pasting into a background tab: ' +
          'put the two chats in two separate Chrome windows.';
      throw new Error('Timed out waiting for Tab ' + role + '. ' + hint);
    }
    await pause(POLL_MS);
  }
}

// One direction of the loop: take "text" and send it from one tab to the other.
async function forward(run, fromRole, toRole, text) {
  checkStop();
  const fromTab = run.roles[fromRole];
  const toTab = run.roles[toRole];
  const settings = run.settings;

  await setStatus(run.statusPrefix + 'sending ' + fromRole + ' → ' + toRole, 'running');
  const baseline = await readTab(toTab);
  const message = Common.wrapMessage(settings.prefix, text, settings.suffix);
  if (!message) throw new Error('There is nothing to send: the last reply was empty.');
  if (message.length > START_TEXT_MAX) {
    throw new Error('The reply is very long (' + message.length + ' characters). ' +
      'Many chat boxes refuse a message that big. Use "Between markers" or "Last code block" ' +
      'to forward a shorter part.');
  }
  await addLog('Sending ' + message.length + ' characters from ' + fromRole + ' to ' + toRole + '.');
  await deliverToTab(toTab, message);

  // The waiter gets the sent text, so a page that shows our message back as
  // the newest "reply" is not mistaken for the AI answering.
  const reply = await waitForNewReply(run, toRole, baseline, message);
  const mode = toRole === 'A' ? settings.copyModeA : settings.copyModeB;
  const picked = applyMode(reply, mode, { start: settings.markerStart, end: settings.markerEnd });
  if (picked.error) {
    throw new Error('I read a new answer from ' + roleTitle(run, toRole) + ' but ' + picked.error);
  }
  await addLog(toRole + ' answered with ' + picked.text.length + ' characters.');
  return picked.text;
}

// Check the stop word and the "identical reply" option. Returns a stop reason.
function checkStopOptions(run, role, text, previousText) {
  const word = String(run.settings.stopWord || '').trim();
  if (word && text.toLowerCase().includes(word.toLowerCase())) {
    return 'Stopped because the reply from ' + role + ' contains "' + word + '".';
  }
  if (run.settings.stopIfSame && previousText && text.trim() === previousText.trim()) {
    return 'Stopped because the new reply from ' + role + ' is identical to the previous one. ' +
      'The two AIs are probably stuck in a loop.';
  }
  return '';
}

// ---------- tab to tab mode ----------
async function runTabMode(run) {
  const settings = run.settings;
  const rounds = Common.effectiveRounds(settings);
  await addLog('Tab mode started: ' + rounds + ' round' + (rounds === 1 ? '' : 's') + '.');

  await setStatus('Reading the first message from Tab A', 'running');
  let outgoing = await readPickedText(run, 'A');
  const previous = { A: outgoing, B: '' };

  for (let round = 1; round <= rounds; round++) {
    run.statusPrefix = 'Round ' + round + '/' + rounds + ': ';
    for (const [from, to] of [['A', 'B'], ['B', 'A']]) {
      checkStop();
      const newText = await forward(run, from, to, outgoing);
      const stopReason = checkStopOptions(run, to, newText, previous[to]);
      previous[to] = newText;
      outgoing = newText;
      if (stopReason) {
        await addLog(stopReason);
        await setStatus(stopReason, 'done');
        return { finished: false, reason: stopReason };
      }
      if (to === 'A') await addLog('Round ' + round + ' of ' + rounds + ' finished.');
    }
  }
  const done = 'Finished all ' + rounds + ' round' + (rounds === 1 ? '' : 's') + '.';
  await setStatus(done, 'done');
  await addLog(done);
  return { finished: true, reason: done };
}

// Keep only the newest few messages, but always start on a user message:
// Gemini requires the conversation to begin with the user's turn.
function trimMemory(memory, limit) {
  const max = Math.max(1, Number(limit) || 1);
  while (memory.length > max) {
    memory.shift();
    if (memory.length && memory[0].role !== 'user') memory.shift();
  }
}

// ---------- API mode: Tab A <-> API ----------
async function runApiMode(run) {
  const settings = run.settings;
  const rounds = Common.effectiveRounds(settings);
  const primary = Common.providerLabel(settings.provider);
  await addLog('API mode started with ' + primary + ' as the main provider, ' + rounds +
    ' round' + (rounds === 1 ? '' : 's') + '.');

  await setStatus('Reading the first message from Tab A', 'running');
  let outgoing = await readPickedText(run, 'A');
  let previous = outgoing;
  const memory = []; // { role: 'user' | 'assistant', content }
  // One cooldown table for the whole run, so a provider that answered 429 in
  // an earlier round is still skipped in the later rounds.
  const cooldowns = { gemini: 0, grok: 0 };

  for (let round = 1; round <= rounds; round++) {
    checkStop();
    run.statusPrefix = 'Round ' + round + '/' + rounds + ': ';

    // 1. Ask the API.
    await setStatus(run.statusPrefix + 'asking ' + primary, 'running');
    memory.push({ role: 'user', content: Common.wrapMessage(settings.prefix, outgoing, settings.suffix) });
    trimMemory(memory, settings.memory);
    const startedAt = Date.now();
    const answer = await ChatApi.requestWithFallback(memory, {
      provider: settings.provider,
      geminiKey: settings.geminiKey,
      geminiModel: settings.geminiModel,
      grokKey: settings.grokKey,
      grokModel: settings.grokModel,
      systemPrompt: settings.systemPrompt
    }, { cooldowns: cooldowns });
    for (const note of answer.notes || []) await addLog(note);
    const providerName = Common.providerLabel(answer.provider);
    await addLog(providerName + ' answered in ' + Math.round((Date.now() - startedAt) / 1000) +
      's with ' + answer.text.length + ' characters.');
    memory.push({ role: 'assistant', content: answer.text });

    // 2. Paste the answer into Tab A and wait for Tab A's own answer.
    await setStatus(run.statusPrefix + providerName + ' answered. Pasting it into Tab A', 'running');
    const baseline = await readTab(run.roles.A);
    const pastedIntoA = Common.wrapMessage(settings.prefix, answer.text, settings.suffix);
    await deliverToTab(run.roles.A, pastedIntoA);
    const reply = await waitForNewReply(run, 'A', baseline, pastedIntoA);
    const picked = applyMode(reply, settings.copyModeA, {
      start: settings.markerStart, end: settings.markerEnd
    });
    if (picked.error) {
      throw new Error('I read a new answer from Tab A but ' + picked.error);
    }
    const stopReason = checkStopOptions(run, 'A', picked.text, previous);
    previous = picked.text;
    outgoing = picked.text;
    if (stopReason) {
      await addLog(stopReason);
      await setStatus(stopReason, 'done');
      return { finished: false, reason: stopReason };
    }
    await addLog('Round ' + round + ' of ' + rounds + ' finished (' + providerName + ').');
  }
  const done = 'Finished all ' + rounds + ' round' + (rounds === 1 ? '' : 's') + '.';
  await setStatus(done, 'done');
  await addLog(done);
  return { finished: true, reason: done };
}

// ---------- checking everything before a run ----------
async function tabInfo(tabId, role) {
  try {
    const tab = await chrome.tabs.get(tabId);
    return { ok: true, tab: tab, key: Common.siteKeyFor(tab.url || ''), label: Common.siteLabel(tab.url || '') };
  } catch (err) {
    // Say which tab and which button, in plain words.
    const which = role === 'A' || role === 'B' ? 'Tab ' + role : 'That tab';
    return {
      ok: false,
      error: which + ' is closed. Open the chat you want and press "Make this ' + which + '" again.'
    };
  }
}

async function validateRun(settings) {
  const roles = await getRoles();
  const problems = [];

  if (!roles.A) problems.push('Tab A is not chosen yet. Open the first chat tab and press "Make this Tab A".');
  if (settings.mode === 'tab' && !roles.B) {
    problems.push('Tab B is not chosen yet. Open the second chat tab and press "Make this Tab B".');
  }
  if (roles.A && roles.B && roles.A === roles.B) {
    problems.push('Tab A and Tab B are the same tab. Use two different tabs.');
  }
  if (problems.length) return { problems: problems };

  // Check every tab that this mode needs, and report a closed one right away.
  const need = settings.mode === 'tab' ? ['A', 'B'] : ['A'];
  for (const role of need) {
    const info = await tabInfo(roles[role], role);
    if (!info.ok) {
      problems.push(info.error);
      continue;
    }
    const stored = await chrome.storage.local.get(Common.pickStorageKey(info.key));
    const picks = stored[Common.pickStorageKey(info.key)] || {};
    const missing = [];
    if (!picks.input) missing.push('chat box');
    if (!picks.reply) missing.push('AI reply');
    if (missing.length) {
      problems.push('Tab ' + role + ' (' + info.label + ') needs the ' + missing.join(' and ') +
        ' picked. Click this tab, then press the pick buttons in the popup.');
    }
  }

  if (settings.mode === 'api') {
    // Only the main provider's key is required. The second key enables the
    // automatic backup, but a run without it still works.
    if (!Common.hasProviderKey(settings, settings.provider)) {
      problems.push('API mode needs an API key for ' + Common.providerLabel(settings.provider) +
        '. Add it in Settings.');
    }
  }
  return { problems: problems };
}

// ---------- starting and stopping ----------
async function startRun() {
  if (activeRun) return { error: 'A run is already going. Press Stop first.' };
  const settings = await getSettings();
  const check = await validateRun(settings);
  if (check.problems.length) {
    // One problem per line, so the popup can show a tidy list.
    const message = check.problems.join('\n');
    await setStatus(message, 'error');
    await addLog(message);
    return { error: message };
  }

  const roles = await getRoles();
  const run = {
    settings: settings,
    roles: roles,
    stopRequested: false,
    statusPrefix: '',
    titles: {},
    heartbeat: null
  };
  for (const role of ['A', 'B']) {
    if (!roles[role]) continue;
    const info = await tabInfo(roles[role], role);
    if (info.ok) run.titles[roles[role]] = info.tab.title || info.label;
  }
  activeRun = run;
  await setRunFlag(true, '');
  await setStatus('Starting...', 'running');
  await addLog('Run started in ' + (settings.mode === 'api' ? 'API' : 'tab to tab') + ' mode.');

  // A small extension call every 10 seconds keeps the service worker awake,
  // even while a long API request is waiting for an answer.
  run.heartbeat = setInterval(() => {
    chrome.runtime.getPlatformInfo().catch(() => {});
  }, HEARTBEAT_MS);

  try {
    const result = settings.mode === 'api' ? await runApiMode(run) : await runTabMode(run);
    return { ok: true, result: result };
  } catch (err) {
    if (err instanceof StopSignal) {
      await setStatus('Stopped by you.', 'stopped');
      await addLog('Run stopped by you.');
      return { ok: true, stopped: true };
    }
    const message = 'Error: ' + (err && err.message ? err.message : String(err));
    await setStatus(message, 'error');
    await addLog(message);
    return { error: message };
  } finally {
    if (run.heartbeat) clearInterval(run.heartbeat);
    // Clear the run BEFORE writing the final status, so the popup sees
    // "running: false" and enables the Start button again.
    activeRun = null;
    await setRunFlag(false, '');
    const last = await chrome.storage.local.get('status');
    if (last.status) {
      await queueWrite(() => chrome.storage.local.set({
        status: Object.assign({}, last.status, { running: false, updatedAt: Date.now() })
      }));
    }
  }
}

async function stopRun() {
  if (!activeRun) {
    await setStatus('Nothing is running.', 'idle');
    await setRunFlag(false, '');
    return { ok: true };
  }
  activeRun.stopRequested = true;
  await setStatus('Stopping...', 'stopped');
  return { ok: true };
}

// ---------- messages from the popup ----------
chrome.runtime.onMessage.addListener((message, sender, sendResponse) => {
  const type = message && message.type;
  (async () => {
    if (type === 'start') return startRun();
    if (type === 'stop') return stopRun();
    if (type === 'state') {
      return {
        running: Boolean(activeRun),
        roles: await getRoles()
      };
    }
    if (type === 'check') {
      const settings = await getSettings();
      return await validateRun(settings);
    }
    return { error: 'Unknown request.' };
  })().then(
    (result) => { try { sendResponse(result); } catch (err) { /* popup closed */ } },
    (err) => {
      try {
        sendResponse({ error: plainError(err, 'Something went wrong in the background worker.') });
      } catch (closeErr) { /* popup closed */ }
    }
  );
  return true; // async answer follows
});

// ---------- startup checks ----------
(async function startup() {
  try {
    const stored = await chrome.storage.local.get('run');
    const run = stored.run || {};
    if (run.running && !activeRun) {
      // Chrome stopped the worker in the middle of a run, so the loop is gone.
````

````js
  function chatBoxFrom(pick) {

  // Selectors that have matched at least one element since this page loaded.
  // It lets the popup tell "this chat has no answer yet" apart from "my pick
  // stopped working", without ever failing a running loop.
  const selectorsThatMatched = new Set();

  function readLastReply(selector) {

        count: 0,

        code: '',

        // which usually means the page was rebuilt or the chat was cleared.
        wasWorking: selectorsThatMatched.has(selector)

    selectorsThatMatched.add(selector);

    // block" really means the last one in the answer that is being read.
    const body = textBodyOf(element);
    const text = Extract.cleanText(Extract.visibleText(body));

      const last = blocks[blocks.length - 1];
      const holder = last.tagName === 'PRE' && last.querySelector('code') ? last.querySelector('code') : last;
      code = Extract.cleanText(Extract.visibleText(holder));
    }
    return { ok: true, count: list.length, text: text, code: code };

  // ---------- pasting and sending ----------
  // React and Vue only notice a value change made through the native setter.
  function setNativeValue(box, text) {
    const proto = box.tagName.toLowerCase() === 'textarea'
      ? window.HTMLTextAreaElement.prototype
      : window.HTMLInputElement.prototype;
    const descriptor = Object.getOwnPropertyDescriptor(proto, 'value');
    if (descriptor && descriptor.set) descriptor.set.call(box, text);
    else box.value = text;
    box.dispatchEvent(new Event('input', { bubbles: true, composed: true }));
    box.dispatchEvent(new Event('change', { bubbles: true }));

  function selectAllContent(box) {
    try {
      const range = document.createRange();
      range.selectNodeContents(box);

      selection.removeAllRanges();

    }

  async function startPicker(kind) {

      mouseX: 0,
      mouseY: 0,
      listeners: []

    say('Picking the <b>' + KIND_NAMES[kind] + '</b>. Hover, press <b>Arrow Up</b> for a bigger area, ' +
      '<b>Arrow Down</b> to go back, <b>click</b> to choose, <b>Esc</b> to cancel.');

      const target = state.target;
      if (!target || !target.isConnected) {
        box.style.display = 'none';
        label.style.display = 'none';
        return;
      }
      const rect = target.getBoundingClientRect();
      box.style.display = 'block';
      box.style.left = rect.left + 'px';
      box.style.top = rect.top + 'px';
      box.style.width = Math.max(0, rect.width) + 'px';
      box.style.height = Math.max(0, rect.height) + 'px';

      label.style.top = (rect.top > 34 ? rect.top - 26 : rect.bottom + 4) + 'px';

      state.base = element;
      state.target = element;
      paint();

      state.mouseX = event.clientX;
      state.mouseY = event.clientY;
      const element = elementUnderPoint(event.clientX, event.clientY);
      if (element && element !== state.base) setBase(element);

      if (event.key === 'Escape') {
        event.preventDefault();

        say('Picking cancelled. Nothing was saved.');
        return;

      if (event.key === 'ArrowUp') {
        event.preventDefault();
        event.stopImmediatePropagation();
        const parent = state.target && state.target.parentElement;
        if (parent && parent !== document.documentElement) {
          state.target = parent;
          paint();
        }
        return;
      }
      if (event.key === 'ArrowDown') {
        event.preventDefault();
        event.stopImmediatePropagation();
        if (state.target && state.base && state.target !== state.base) {
          let child = state.base;
          while (child.parentElement && child.parentElement !== state.target) child = child.parentElement;
          state.target = child.parentElement === state.target ? child : state.base;
          paint();
        }
        return;
      }
      // Every other key belongs to the picker, not to the page.
      event.preventDefault();
      event.stopImmediatePropagation();

    async function onClick(event) {
      event.preventDefault();
      event.stopImmediatePropagation();
      if (!state.target) return;
      const rect = state.target.getBoundingClientRect();
      const point = {
        x: Math.round(state.mouseX || rect.left + rect.width / 2),
        y: Math.round(state.mouseY || rect.top + rect.height / 2)
      };
      for (const [type, handler] of state.listeners) window.removeEventListener(type, handler, true);
      state.listeners = [];
      await savePick(key, kind, state.target, point, say);
    }

    // Anything else the page could react to is swallowed in the capture phase.
    const swallow = (event) => {
      event.preventDefault();
      event.stopImmediatePropagation();
    };

    const add = (type, handler) => {
      window.addEventListener(type, handler, true);
      state.listeners.push([type, handler]);
    };
    add('mousemove', onMouseMove);
    add('keydown', onKeyDown);
    add('click', onClick);
    for (const type of ['mousedown', 'mouseup', 'pointerdown', 'pointerup', 'dblclick', 'auxclick', 'contextmenu', 'wheel']) {
      add(type, type === 'wheel' ? () => {} : swallow);
    }

    // Start with the element under the mouse, or the middle of the page.
    const first = elementUnderPoint(state.mouseX || window.innerWidth / 2, state.mouseY || window.innerHeight / 2);
    if (first) setBase(first);
  }

  // Build and save the selector for the element the user clicked.
  async function savePick(key, kind, clicked, point, say) {
    let element = clicked;

    // Chat box and send button: use the nearest real control.
    if (kind === 'input') {
      element = clicked.closest('textarea, input, [contenteditable="true"], [role="textbox"]') || clicked;
    } else if (kind === 'send') {
      element = clicked.closest('button, [role="button"], a') || clicked;
    }

    const result = Selectors.buildSelector(element, kind, document);
    if (result.error) {
      // Stay in pick mode so the user can try another spot.
      say('<b>I could not save that spot.</b> ' + result.error + ' Press Esc to cancel.');
      await startPickerFresh(kind);
      return;
    }

    const name = PICK_KEY_PREFIX + key;
    const stored = (await chrome.storage.local.get(name))[name] || {};
    stored[kind] = {
      selector: result.selector,
      count: result.count,
      kind: kind,
      pickedAt: Date.now(),
      x: point.x,
      y: point.y
    };
    await chrome.storage.local.set({ [name]: stored });

    let note = '';
    if (kind === 'reply') {
      note = result.count === 1
        ? ' It matches 1 reply now; new replies will be found automatically.'
        : ' It matches ' + result.count + ' replies and the last one is always the newest.';
      // If the selector also matches the user's own messages, say so, because
      // the newest match could then be a question instead of an answer.
      const share = Selectors.aiShareOf(result.selector, document);
      if (result.count > 1 && share < 0.6) {
        note += ' Warning: some of the matches look like your own messages. ' +
          'Try picking the reply again and press Arrow Up so only the AI answers match.';
      }
      note += ' If that is wrong, pick it again and press Arrow Up before clicking.';
    } else if (result.fallback) {
      note = ' I used a long structural selector because no stable attributes were found; ' +
        'it may need refreshing after the site is redesigned.';
    } else if (result.exact === false) {
      note = ' Warning: this spot does not match exactly one element, so it is less reliable.';
    }
    say('<b>Saved the ' + KIND_NAMES[kind] + ' (' + result.count +
      (result.count === 1 ? ' match' : ' matches') + ').</b>' + note + ' Picking is finished.');
    await sleep(1800);
    stopPicker();
  }

      }

      return 'the extension was reloaded. Please refresh this page (F5) and try again.';
    }
    return text;
````

````js
      if (!weight) continue;

    }
    return null;
````

// Works on a real page element (tab to tab mode) and on plain text (API mode).
// This file is plain JavaScript with no dependencies, so it can be loaded in
// a content script, in the background service worker, or in Node for tests.

  'use strict';

  const MODES = { FULL: 'full', CODE: 'code', MARKERS: 'markers' };
  const MODE_LABELS = {
    full: 'Full reply',
    code: 'Last code block',
    markers: 'Between markers'
  };

  // Make text consistent: Windows line endings are changed to normal newlines.
  function cleanText(value) {
    return String(value == null ? '' : value).replace(/\r\n?/g, '\n').trim();
  }

  // Read text that the user can actually see. innerText respects line breaks
  // and hidden elements; textContent is the fallback in simple test pages.
  function visibleText(el) {
    if (!el) return '';
    const inner = el.innerText;
    if (typeof inner === 'string' && inner.trim() !== '') return inner;
    return el.textContent || '';

    const start = (markers && String(markers.start).trim()) || 'START';
    const end = (markers && String(markers.end).trim()) || 'END';
    return { start: start, end: end };
  }

  // Text between the LAST start marker and the first end marker after it.
  function betweenMarkers(text, markers) {
    const pair = markersOf(markers);
    const source = String(text == null ? '' : text);
    const startAt = source.lastIndexOf(pair.start);
    if (startAt === -1) {
      return {
        error: 'I could not find the START marker "' + pair.start + '" in the last reply. ' +
          'Tell the AI to put that word where the part you want to forward begins.'
      };
    }
    const afterStart = startAt + pair.start.length;
    const endAt = source.indexOf(pair.end, afterStart);
    if (endAt === -1) {
      return {
        error: 'I found the START marker but no END marker "' + pair.end + '" after it in the last reply.'
      };
    }
    const value = cleanText(source.slice(afterStart, endAt));
    if (!value) return { error: 'The text between the START and END markers is empty.' };
    return { text: value };
  }

  // Last fenced code block in plain text (API answers arrive as markdown).
  function lastCodeFromText(text) {
    const source = String(text == null ? '' : text);
    const re = /```[^\n`]*\n?([\s\S]*?)```/g;
    let match;
    let last = null;
    while ((match = re.exec(source)) !== null) last = match[1];
    if (last === null) return { error: 'No code block found in the last reply.' };
    const value = cleanText(last);
    if (!value) return { error: 'The last code block in the reply is empty.' };
    return { text: value };
  }

  // Last <pre><code> (or <pre>) inside the reply element.
  function lastCodeFromElement(el) {
    if (!el) return { error: 'No code block found in the last reply.' };
    const blocks = el.querySelectorAll('pre code, pre');
    if (!blocks.length) return { error: 'No code block found in the last reply.' };
    const last = blocks[blocks.length - 1];
    const holder = last.tagName === 'PRE' && last.querySelector('code') ? last.querySelector('code') : last;
    const value = cleanText(visibleText(holder));
    if (!value) return { error: 'The last code block in the reply is empty.' };
    return { text: value };

  function applyModeToText(text, mode, markers) {
    const full = cleanText(text);
    if (!full) return { error: 'The reply is empty.' };
    if (mode === MODES.CODE) return lastCodeFromText(full);
    if (mode === MODES.MARKERS) return betweenMarkers(full, markers);

    if (mode === MODES.MARKERS) return betweenMarkers(full, markers);

  // on top of the "seconds of silence" setting.

````js
// with automatic fallback when one of them fails.
// Plain JavaScript, no dependencies. Works in the background worker and in Node
// (for tests) through the small wrapper at the top.
//
// Provider details (checked against the official docs, November 2026):
//   Gemini: POST https://generativelanguage.googleapis.com/v1beta/models/{model}:generateContent
//           header: x-goog-api-key
//           body:   { systemInstruction: { parts: [{ text }] },
//                     contents: [{ role: "user"|"model", parts: [{ text }] }] }
//           answer: candidates[0].content.parts[].text
//   Grok:   POST https://api.x.ai/v1/chat/completions   (OpenAI compatible)
//           header: Authorization: Bearer <key>
//           body:   { model, messages: [{ role, content }] }
//           answer: choices[0].message.content
(function (root, factory) {
  'use strict';
  const api = factory();
  if (typeof module === 'object' && module.exports) module.exports = api;
  if (root) root.ChatBridgeApi = api;
})(typeof self !== 'undefined' ? self : null, function () {
  'use strict';

  const GEMINI_BASE = 'https://generativelanguage.googleapis.com/v1beta/models/';
  const GROK_URL = 'https://api.x.ai/v1/chat/completions';
  const REQUEST_TIMEOUT_MS = 120000; // an API answer may take a while
  const COOLDOWN_MS = 60000;         // a rate limited provider rests 60 seconds
  const RETRY_WAIT_MS = 2000;        // wait before the single retry

  // When each provider may be used again (a timestamp per provider).
  // It lives at module level so a 429 in one round is remembered in the next.
  // A run can pass its own object in deps.cooldowns to keep it per run.
  const sharedCooldowns = { gemini: 0, grok: 0 };

  // Read the cooldown table: the one from the caller when given, else the
  // shared one. Missing entries are created, so callers can pass {}.
  function cooldownsFrom(deps) {
    const table = deps && deps.cooldowns;
    if (!table || typeof table !== 'object') return sharedCooldowns;
    if (typeof table.gemini !== 'number') table.gemini = 0;
    if (typeof table.grok !== 'number') table.grok = 0;
    return table;
  }

  function providerLabel(provider) {
    return provider === 'grok' ? 'Grok' : 'Gemini';
  }

  function makeError(message, extra) {
    return Object.assign(new Error(message), extra || {});
  }

  // One HTTP call, with a timeout so a stuck request cannot freeze the loop.
  async function postJson(fetchFn, url, headers, body) {
    const controller = typeof AbortController === 'function' ? new AbortController() : null;
    const timer = controller ? setTimeout(() => controller.abort(), REQUEST_TIMEOUT_MS) : null;
    try {
      const response = await fetchFn(url, {
        method: 'POST',
        headers: Object.assign({ 'Content-Type': 'application/json' }, headers),
        body: JSON.stringify(body),
        signal: controller ? controller.signal : undefined
      });
      let data = null;
      let text = '';
      try {
        text = await response.text();
        data = text ? JSON.parse(text) : null;
      } catch (err) {
        data = null;
      }
      return { status: response.status, ok: response.ok, data: data, text: text };
    } finally {
      if (timer) clearTimeout(timer);
    }
  }

  // Turn an HTTP status into a short, plain explanation.
  function explainStatus(status) {
    if (status === 429) return 'the free quota or rate limit is used up';
    if (status === 401 || status === 403) return 'the API key was rejected';
    if (status === 404) return 'the model name was not found';
    if (status === 400) return 'the request was refused';
    if (status >= 500) return 'the service is having problems';
    return 'the server refused the request';
  }

  // Read the answer text out of a Gemini response. Throws if there is no text.
  function readGeminiText(data) {
    const candidate = data && data.candidates && data.candidates[0];
    const parts = (candidate && candidate.content && candidate.content.parts) || [];
    const text = parts.map((part) => (part && typeof part.text === 'string' ? part.text : '')).join('').trim();
    if (text) return text;
    const blocked = data && data.promptFeedback && data.promptFeedback.blockReason;
    if (blocked) throw makeError('Gemini refused to answer (reason: ' + blocked + ').', { kind: 'empty' });
    const reason = candidate && candidate.finishReason;
    throw makeError('Gemini sent no text back' + (reason ? ' (finish reason: ' + reason + ')' : '') + '.',
      { kind: 'empty' });
  }

  // Read the answer text out of a Grok (OpenAI style) response.
  function readGrokText(data) {
    const choice = data && data.choices && data.choices[0];
    const content = choice && choice.message && choice.message.content;
    if (typeof content === 'string' && content.trim()) return content.trim();
    const reason = choice && choice.finish_reason;
    throw makeError('Grok sent no text back' + (reason ? ' (finish reason: ' + reason + ')' : '') + '.',
      { kind: 'empty' });
  }

  function normalizeMessages(messages) {
    if (!Array.isArray(messages)) return [];
    return messages
      .filter((m) => m && (m.role === 'user' || m.role === 'assistant') && String(m.content || '').trim())
      .map((m) => ({ role: m.role, content: String(m.content) }));
  }

  // Gemini request body. Gemini calls the assistant role "model".
  function buildGeminiRequest(model, messages, systemPrompt) {
    const contents = normalizeMessages(messages).map((m) => ({
      role: m.role === 'assistant' ? 'model' : 'user',
      parts: [{ text: m.content }]
    }));
    const body = { contents: contents };
    if (systemPrompt && String(systemPrompt).trim()) {
      body.systemInstruction = { parts: [{ text: String(systemPrompt) }] };
    }
    const safeModel = String(model || 'gemini-2.5-flash').trim().replace(/^models\//, '');
    return { url: GEMINI_BASE + encodeURIComponent(safeModel) + ':generateContent', body: body };
  }

  // Grok request body, in the OpenAI chat format.
  function buildGrokRequest(model, messages, systemPrompt) {
    const list = [];
    if (systemPrompt && String(systemPrompt).trim()) {
      list.push({ role: 'system', content: String(systemPrompt) });
    }
    for (const m of normalizeMessages(messages)) list.push({ role: m.role, content: m.content });
    return {
      url: GROK_URL,
      body: { model: String(model || 'grok-4').trim(), messages: list, stream: false }
    };
  }

  // Short wait that the tests can switch off.
  function wait(ms, deps) {
    const delay = deps && typeof deps.sleep === 'function'
      ? deps.sleep
      : (t) => new Promise((resolve) => setTimeout(resolve, t));
    return deps && deps.retryWaitMs === 0 ? Promise.resolve() : delay(ms);
  }

  // One turn of an API chat with automatic fallback.
  //   messages : the conversation so far; the last item is the new question
  //   settings : { provider, geminiKey, geminiModel, grokKey, grokModel, systemPrompt }
  //   deps     : { fetch, now, sleep, cooldownMs, retryWaitMs }  (all optional)
  // Returns { text, provider, fellBack, notes }.
  // The notes are plain-English lines for the popup log.
  // Throws a plain Error if both providers fail.
  async function requestWithFallback(messages, settings, deps) {
    const d = deps || {};
    const fetchFn = d.fetch || (typeof fetch === 'function' ? fetch.bind(globalThis) : null);
    const now = typeof d.now === 'function' ? d.now : Date.now;
    const cooldownMs = typeof d.cooldownMs === 'number' ? d.cooldownMs : COOLDOWN_MS;
    if (!fetchFn) throw new Error('This browser cannot contact the API. Please use Chrome.');

    const primary = settings.provider === 'grok' ? 'grok' : 'gemini';
    const secondary = primary === 'grok' ? 'gemini' : 'grok';
    const notes = [];
    const keys = {
      gemini: String(settings.geminiKey || '').trim(),
      grok: String(settings.grokKey || '').trim()
    };
    const models = { gemini: settings.geminiModel, grok: settings.grokModel };

    // Providers that hit a 429 rest for 60 seconds before being used again.
    // This table is NOT created here: it outlives a single request, so the
    // second round already knows that Gemini is resting.
    const cooldownUntil = cooldownsFrom(d);

    // One HTTP attempt with one provider. Throws a plain Error on any problem.
    async function attempt(provider) {
      const label = providerLabel(provider);
      const isGemini = provider === 'gemini';
      const req = isGemini
        ? buildGeminiRequest(models.gemini, messages, settings.systemPrompt)
        : buildGrokRequest(models.grok, messages, settings.systemPrompt);
      const headers = isGemini ? { 'x-goog-api-key': keys[provider] } : { Authorization: 'Bearer ' + keys[provider] };
      let res;
      try {
        res = await postJson(fetchFn, req.url, headers, req.body);
      } catch (err) {
        const timedOut = err && err.name === 'AbortError';
        throw makeError(label + (timedOut ? ' timed out after 2 minutes.' : ' could not be reached (network problem).'),
          { kind: 'network' });
      }
      if (!res.ok) {
        throw makeError(label + ' could not answer: ' + explainStatus(res.status) + ' (HTTP ' + res.status + ').',
          { kind: 'http', status: res.status });
      }
      return isGemini ? readGeminiText(res.data) : readGrokText(res.data);
    }

    // Use a provider: once, and once more after 2 seconds if it failed.
    async function tryProvider(provider) {
      const label = providerLabel(provider);
      if (now() < cooldownUntil[provider]) {
        const seconds = Math.ceil((cooldownUntil[provider] - now()) / 1000);
        throw makeError(label + ' is resting after a rate limit (' + seconds + 's left).', { kind: 'cooldown' });
      }
      if (!keys[provider]) {
        throw makeError('No API key saved for ' + label + '.', { kind: 'nokey' });
      }
      let lastError = null;
      for (let attemptNo = 1; attemptNo <= 2; attemptNo++) {
        try {
          const text = await attempt(provider);
          if (text) return text;
          lastError = makeError(label + ' sent an empty answer.', { kind: 'empty' });
        } catch (err) {
          lastError = err;
        }
        if (lastError.kind === 'nokey') break;
        if (lastError.status === 429) {
          cooldownUntil[provider] = now() + cooldownMs;
          notes.push(label + ' hit the rate limit, so it will rest 60 seconds.');
          break; // a rate limit will not clear by retrying right away
        }
        if (attemptNo === 1) {
          notes.push(lastError.message + ' Trying once more...');
          await wait(RETRY_WAIT_MS, d);
        }
      }
      throw lastError || makeError(label + ' did not answer.', { kind: 'empty' });
    }

    // First choice.
    try {
      const text = await tryProvider(primary);
      return { text: text, provider: primary, fellBack: false, notes: notes };
    } catch (err) {
      if (err.kind === 'nokey' && !keys[secondary]) {
        throw new Error('No API key saved for ' + providerLabel(primary) +
          '. Please add your API key in the Settings section.');
      }
      notes.push(err.message);
      notes.push('Switching to the backup provider (' + providerLabel(secondary) + ').');
    }

    // Second choice.
    try {
      const text = await tryProvider(secondary);
      notes.push('The answer came from ' + providerLabel(secondary) + ' instead.');
      return { text: text, provider: secondary, fellBack: true, notes: notes };
    } catch (err) {
      throw new Error('Both API providers failed. ' + notes.join(' ') + ' ' + err.message);
    }
  }

  return {
    COOLDOWN_MS,
    GEMINI_BASE,
    sharedCooldowns,
    GROK_URL,
    REQUEST_TIMEOUT_MS,
    RETRY_WAIT_MS,
    buildGeminiRequest,
    buildGrokRequest,
    explainStatus,
    providerLabel,
    requestWithFallback
  };
});
````

````js
function tabMessage(tabId, message) {
  return new Promise((resolve) => {
    try {
      chrome.tabs.sendMessage(tabId, message, (reply) => {
        if (chrome.runtime.lastError) {
          resolve({ error: chrome.runtime.lastError.message });
          return;
        }
        resolve(reply || { error: 'The page did not answer.' });
      });
    } catch (err) {
      resolve({ error: err && err.message ? err.message : 'The page did not answer.' });
    }
  });
}

function bgMessage(message) {
  return new Promise((resolve) => {
    chrome.runtime.sendMessage(message, (reply) => {
      if (chrome.runtime.lastError) {
        resolve({ error: chrome.runtime.lastError.message });
        return;
      }
      resolve(reply || {});
    });
  });
}

// The content script may be missing on tabs opened before the extension was
// installed or reloaded. Add it, then try again.
async function tabMessageWithRetry(tabId, message) {
  const first = await tabMessage(tabId, message);
  if (!first || !first.error) return first;
  const missing = /Receiving end does not exist|Could not establish connection|message port closed/i.test(first.error);
  if (!missing) return first;
  try {
    await chrome.scripting.executeScript({
      target: { tabId: tabId },
      files: ['selectors.js', 'extract.js', 'common.js', 'content.js']
    });
  } catch (err) {
    return { error: 'This page cannot be used by the extension. Open a normal website tab.' };
  }
  return await tabMessage(tabId, message);
}

// The worker answers "start" only when the whole run has finished, so the
// popup does not wait for it: the live status line shows the progress instead.
// A run that cannot start answers immediately with an error, which we show.
function start() {
  if (busy) return;
  busy = true;
  el.start.disabled = true;
  flushSave().then(() => {
    el.start.disabled = true;
    chrome.runtime.sendMessage({ type: 'start' }, (reply) => {
      busy = false;
      if (chrome.runtime.lastError) {
        setTestOutput('Cannot start yet', chrome.runtime.lastError.message, true);
        showStatus(state.status);
        return;
      }
      if (reply && reply.error) {
        setTestOutput('Cannot start yet', reply.error, true);
      }
      // The status line was updated live while the run went on.
      showStatus(state.status);
    });
  });
}

// Live updates from the background worker.
chrome.storage.onChanged.addListener((changes, area) => {
  if (area !== 'local') return;
  if (changes.status) {
    state.status = changes.status.newValue;
    showStatus(state.status);
  }
  if (changes.log) {
    state.log = changes.log.newValue || [];
    renderLog(state.log);
  }
  if (changes.roles) {
    state.roles = changes.roles.newValue || { A: null, B: null };
    renderRoles();
  }
  // A pick was saved or cleared for the site of this tab: refresh the ticks.
  if (currentKey && changes[Common.pickStorageKey(currentKey)]) {
    currentPicks = changes[Common.pickStorageKey(currentKey)].newValue || null;
    renderPicks();
  }

  // Settings changed in another window: update the form, but never while the
  // user is typing in it (that would move the cursor around).
  if (changes.settings) {
    const next = Common.normalizeSettings(changes.settings.newValue);
    settings = next;

    if (idle && !saveTimer) fillForm(next);
    renderModeBadge();

});
````

````html
  <h2>1. Pick the three spots</h2>
  <ol>
    <li>Click the Chat Bridge icon while the chat tab is active.</li>
    <li>Press <strong>Pick chat box</strong>, then click inside the page's message box.</li>
    <li>Press <strong>Pick send button</strong>, then click the send arrow.</li>

  <h2>4. Waiting until the AI is finished</h2>
  <p>Chat Bridge reads the newest answer once a second. The answer counts as started when
    the text (or the number of replies) changes, and as finished when it has not changed for
    the <strong>seconds of silence</strong> you set.</p>

    <strong>20–30</strong>. Agents go quiet while they run tools, and a low value would
    forward a half-finished answer.
  </div>

    <li><strong>Stop</strong> — stops immediately, also in the middle of a wait.</li>
  </ul>

    answered.</p>
````
